**Goal:** Tapping a card's fatura in Contas a pagar opens `Fatura {nome}`. It shows what weighs on the fatura (Assinaturas, Parcelas, Compras, largest first) and the next six faturas already committed.

**Architecture:** One read endpoint in the months slice, `GET /api/months/{month}/faturas/{card_id}`. It reuses `faturas()` (D28/D30a) and the existing `GROUP BY date` query, so the numbers match Mês. Web: a pure grouping function and a `FaturaSheet` feature. The `bills-of-month` widget opens the sheet and lends it `TransactionSheet` for the row tap.

**Tech Stack:** axum 0.8 + sqlx + rust_decimal; Next.js 16 static export, TanStack Query, openapi-fetch, Vitest, Playwright.

**Spec:** `docs/specs/cards/SOL-119.md` (brief, copy, states, technical design). Worktree `/Users/emerson/Documents/workspace/open-source/soldi-119`, branch `feat/SOL-119-card-panel`.

/// SOL-119: a card's fatura due in `month`, what fills it, and the next six months' committed totals.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, ToSchema)]
pub struct CardFatura {
    /// `YYYY-MM`, the fatura's due month.
    pub month: String,
    /// Spends plus the D30a carry: what Contas a pagar shows unpaid.
    #[serde(serialize_with = "crate::money")]
    pub total: Decimal,
    /// Left over from last month's partial payment (D30a); 0 otherwise.
    #[serde(serialize_with = "crate::money")]
    pub carried: Decimal,
    /// The card's payments dated in `month`; 0 when unpaid.
    #[serde(serialize_with = "crate::money")]
    pub paid: Decimal,
    /// The fatura's spends, newest date first.
    pub entries: Vec<Transaction>,
    /// The next six months' faturas from rows already written (D16), zero included.
    pub ahead: Vec<FaturaAhead>,
}

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, ToSchema)]
pub struct FaturaAhead {
    /// `YYYY-MM`.
    pub month: String,
    #[serde(serialize_with = "crate::money")]
    pub total: Decimal,
}

/// How many faturas after the viewed one the card panel lists (SOL-119).
pub const FATURAS_AHEAD: u32 = 6;

pub fn check_opening(month: NaiveDate, opening: NaiveDate) -> Result<(), AppError> {
    if month < first_of(opening) {
        return Err(AppError::rule("month-before-opening", "The month is before the opening date."));
    }
    Ok(())
}

/// SOL-119: `card`'s fatura due in `month` (its first day), from the shared `faturas` map.
pub fn card_fatura(
    card: &MonthBill,
    month: NaiveDate,
    faturas: &Faturas,
    payments: &[Payment],
    entries: Vec<Transaction>,
) -> CardFatura {
    let total_in = |month: NaiveDate| faturas.get(&(card.id, month)).copied().unwrap_or_default();
    let paid_in = |month: NaiveDate| -> Decimal {
        payments
            .iter()
            .filter(|payment| payment.bill_id == card.id && first_of(payment.date) == month)
            .map(|payment| payment.amount)
            .sum()
    };
    // D30a, as `faturas` does: only a partial payment carries; an unpaid month is its own D17 row.
    let carried = month
        .checked_sub_months(Months::new(1))
        .map_or(Decimal::ZERO, |last| match paid_in(last) {
            paid if paid > Decimal::ZERO => (total_in(last) - paid).max(Decimal::ZERO),
            _ => Decimal::ZERO,
        });
    CardFatura {
        month: month.format("%Y-%m").to_string(),
        total: total_in(month),
        carried,
        paid: paid_in(month),
        entries: entries
            .into_iter()
            .filter(|entry| {
                card.closing_day.is_some_and(|closing| {
                    first_of(fatura_due(entry.date, closing, card.due_day)) == month
                })
            })
            .collect(),
        ahead: (1..=FATURAS_AHEAD)
            .filter_map(|n| month.checked_add_months(Months::new(n)))
            .map(|ahead| FaturaAhead {
                month: ahead.format("%Y-%m").to_string(),
                total: total_in(ahead),
            })
            .collect(),
    }
}

```rust
/// SOL-119: a card's fatura and the next six; 5 queries after D36 growth.
pub async fn fatura(pool: &PgPool, user: &CurrentUser, month: &str, card_id: Uuid) -> Result<CardFatura, AppError> {
    let month = domain::parse_month(month)?;
    let last = month
        .checked_add_months(Months::new(domain::FATURAS_AHEAD))
        .map_or(month, domain::month_end);
    let until = transactions::domain::open_until(crate::today_sao_paulo(), last);
    transactions::service::extend_open_series(pool, user, until).await?;
    let mut conn = pool.acquire().await?;
    let opening = settings::repo::find(&mut conn, user.id).await?.opening_date;
    domain::check_opening(month, opening)?;
    let bills = repo::bills(&mut conn, user.id).await?;
    let card = bills
        .iter()
        .find(|bill| bill.id == card_id && bill.closing_day.is_some())
        .ok_or(AppError::NotFound)?;
    let sums = repo::day_sums(&mut conn, user.id, opening, last, domain::cards_from(opening)).await?;
    let payments = repo::payments(&mut conn, user.id, domain::first_of(opening), last).await?;
    let from = month.checked_sub_months(Months::new(2)).unwrap_or(month);
    let entries = transactions::repo::list(&mut conn, user.id, from, domain::month_end(month), None, Some(card_id)).await?;
    let faturas = domain::faturas(&bills, &sums, &payments);
    Ok(domain::card_fatura(card, month, &faturas, &payments, entries))
}

const LABELS: Label[] = ["Assinaturas", "Parcelas", "Compras"];
const PER_YEAR = { month: 12, week: 52, year: 1 } as const;

const labelOf = ({ series }: Transaction): Label => {
  if (!series) return "Compras";
  return series.open_ended ? "Assinaturas" : "Parcelas";
};

/** `members` come newest date first, as the API lists them. */
const rowOf = (key: string, members: Transaction[], money: (amount: string) => string): FaturaRow => {
  const [latest] = members;
  const { series } = latest;
  let caption = shortDay(latest.date);
  if (series?.open_ended) {
    caption =
      series.every === "year"
        ? OPEN_LABEL.year
        : `${OPEN_LABEL[series.every]} · ${money(fromCents(cents(latest.amount) * PER_YEAR[series.every]))} por ano`;
    if (members.length > 1) caption += ` · ${members.length} nesta fatura`;
  } else if (series) {
    caption =
      series.position === series.total
        ? `${series.total}/${series.total} · última`
        : `${series.position}/${series.total} · faltam ${series.total - series.position}`;
  }
  const amount = fromCents(members.reduce((sum, member) => sum + cents(member.amount), 0));
  return { key, title: latest.note || latest.category.name, caption, amount, entry: latest };
};

/** SOL-119: what weighs on a fatura, in a fixed order, largest first; a series is one row. */
export const faturaGroups = (entries: Transaction[], total: string, money: (amount: string) => string): FaturaGroup[] => {
  const bySeries = new Map<string, Transaction[]>();
  for (const entry of entries) {
    const key = entry.series?.id ?? entry.id;
    bySeries.set(key, [...(bySeries.get(key) ?? []), entry]);
  }
  const base = cents(total);
  return LABELS.flatMap((label) => {
    const rows = [...bySeries]
      .filter(([, members]) => labelOf(members[0]) === label)
      .map(([key, members]) => rowOf(key, members, money))
      .sort((a, b) => cents(b.amount) - cents(a.amount) || a.title.localeCompare(b.title, "pt-BR"));
    if (rows.length === 0) return [];
    const sum = rows.reduce((acc, row) => acc + cents(row.amount), 0);
    return [{ label, total: fromCents(sum), share: base > 0 ? wholeShare(sum, base) : 0, rows }];
  });
};
```
